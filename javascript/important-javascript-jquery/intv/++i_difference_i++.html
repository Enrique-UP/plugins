// Pre-increment
            let a = 5;
            let b = ++a; // यहाँ पहले i की वैल्यू 6 हुई, फिर b में 6 असाइन हुई
            console.log(a); // 6
            console.log(b); // 6
        
            // Post-increment
            let i = 5;
            let c = i++; // यहाँ पहले c में i की पुरानी वैल्यू (5) असाइन हुई, फिर i की वैल्यू बढ़कर 6 हुई
            console.log(i); // 6
            console.log(c); // 5
