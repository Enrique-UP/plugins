The call(), apply(), bind() Method

    
        const obj1 = {
            name: "Enrique",
            age: 40,
            place: "Delhi",
            full: function (street, state) {
                return `Hi, I am ${this.name} from ${this.place} (${street}, ${state}), aged ${this.age} years.`;
            }
        };
        const obj2 = {
            name:"Rohit",
            age:50,
            place:"Mumbai"
        }
        console.log(obj1.full.call(obj2, 25, "Haryana")); // Hi, I am Rohit from Mumbai (25, Haryana), aged 50 years.
        console.log(obj1.full.apply(obj2, [25, "Haryana"])); // Hi, I am Rohit from Mumbai (25, Haryana), aged 50 years.
        console.log(obj1.full.bind(obj2, 50, "Nagaland")()); // Hi, I am Rohit from Mumbai (50, Nagaland), aged 50 years.
