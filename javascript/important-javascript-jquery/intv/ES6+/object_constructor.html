    // 1. Ek Blueprint banaya (Constructor Function)
      function Mobile(brand) {
          this.brand = brand;
      }

      // Is Blueprint ke 'prototype' mein ek feature daal diya ki sabhi mobile call kar sakte hain
      Mobile.prototype.makeCall = function() {
          console.log(`${this.brand} se call ho rahi hai...`);
      };

      // 2. Ek naya mobile phone banaya (Object)
      const myPhone = new Mobile('Samsung');

      // myPhone ke paas apna khud ka 'makeCall' function nahi hai, 
      // lekin wo prototype ki wajah se isko use kar sakta hai!
      myPhone.makeCall(); // Output: Samsung se call ho rahi hai...


      // 3. Ab fark dekhiye:
      // 'myPhone' ek object hai, toh iske paas '__proto__' hoga jo connection dikhata hai
      console.log(myPhone.__proto__ === Mobile.prototype); // true (Yani myPhone, Mobile blueprint se juda hai)
