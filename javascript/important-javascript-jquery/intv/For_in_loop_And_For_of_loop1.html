(function(){
			const details = ["Enrique", 34, "Uttar Pradesh"];
			for(let i in details){
				console.log(i, details[i]);
			}
			// 0 Enrique
			// 1 34
			// 2 Uttar Pradesh
			
			console.log("-------------------------------------------------------------------");
			
			for(let i of details){
				console.log(i, details[i]);
			}
			// Enrique undefined
			// 34 undefined
			// Uttar Pradesh undefined
			
			console.log("-------------------------------------------------------------------");
			
			for(let i of details.keys()){
				console.log(i, details[i]);
			}
			// 0 Enrique
			// 1 34
			// 2 Uttar Pradesh
			
			
			
			console.log("--------------------------------------------------------------------------------------------------------------------------------------");
			
			
			
			
			const details1 = {
				name:"Enrique",
				age:34,
				place:"Uttar Pradesh"
			};
			for(let i in details1){
				console.log(i, details1[i]);
			}
			// name Enrique
			// age 34
			// place Uttar Pradesh
			
			console.log("-------------------------------------------------------------------");
			
			for (let [key, value] of Object.entries(details1)) {
				console.log(key, value);
			}
			// name Enrique
			// age 34
			// place Uttar Pradesh
			
			
			
			console.log("--------------------------------------------------------------------------------------------------------------------------------------");
		})();
